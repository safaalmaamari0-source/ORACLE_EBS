====CREATE A PROCEDURE====

SQL>CREATE OR REPLACE PROCEDURE SAFA_PR_TEST IS
   BEGIN
     DBMS_OUTPUT.PUT_LINE('WELCOM TO ORACLE PROC..!');
   END;
/

--Call the procedure:

SQL>EXEC SAFA_PR_TEST;

--OR

SQL>BEGIN
     SAFA_PR_TEST;
    END;
/

====CREATE A FUNCTION===

SQL>CREATE OR REPLACE FUNCTION SAFA_FN_TEST RETURN VARCHAR2 IS 
    V_MSG VARCHAR2(2000);
    BEGIN
      V_MSG := 'WELCOME TO ORACLE FUNCTIONS...!';
    END;
/

--TO RUN THE FUNCTION 

SQL>SELECT SAFA_FN_TEST FROM DUAL;


--CALL THE PROCEDURE AND FUNCTION IN ONE BLOCK:

SQL> DECLER 
     X VARCHAR2(2000);
     BEGIN
       SAFA_PR_TEST;
       X := SAFA_FN_TEST;
       DBMS_OUTPUT.PUT_LINE(X);
      END;
/
